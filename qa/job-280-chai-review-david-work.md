## ฐานข้อมูล — คำสั่งอ่านเพื่อตรวจ (อ่านเท่านั้น)

```sql
-- 1) สถานะ QS822N (ต้องเห็น active = t ทั้ง 2 ตาราง และ 50+ = 8075)
select 'installed', code, price_per_sqm_thb, active, aliases::text from installed_stone_prices where code='QS822N'
union all
select 'sheet', code, price_50_plus_thb, active, aliases::text from sheet_stone_prices where code='QS822N';
-- 2) ต้องไม่มีรหัส/alias QS288 เหลือทั้งฐาน
select 'installed', count(*) from installed_stone_prices where aliases @> ARRAY['QS288'] or aliases @> ARRAY['QS 288'] or code='QS288'
union all select 'sheet', count(*) from sheet_stone_prices where aliases @> ARRAY['QS288'] or aliases @> ARRAY['QS 288'] or code='QS288';
-- 3) อ่าง 6 แถวที่ถูกแก้ (ชื่อ/รหัสต้องตรงแคตตาล็อก) + อ่างที่ยังอ้างสีที่ถูกซ่อน
select b.sku, b.color_code, b.color_name, b.active,
       (select i.name from installed_stone_prices i where i.code = b.color_code) as catalogue_name
from basin_prices b where b.sku in ('KF004','KF005','KF008','KF010','KF012','KF013','KF024') order by b.sku;
-- 4) นับรวม (อ้างอิง): sheet active 64 · installed active 65 · สีที่ซ่อน 12 · อ่าง 30 แถว (KF024 ยัง active ในตารางอ่าง — เดวิดตั้งใจไม่แตะ)
```
